# Windows version resource for the console executable (psditool.exe).
#
# Windows shows this in the file properties dialog, and SmartScreen and most
# corporate software inventories read it too. An executable with no version
# resource looks anonymous, which is exactly the wrong impression for a tool
# that gets copied onto operational machines.
#
# filevers and prodvers must be 4-integer tuples; the readable strings below
# are what users actually see.

VSVersionInfo(
    ffi=FixedFileInfo(
        filevers=(2, 5, 0, 0),
        prodvers=(2, 5, 0, 0),
        mask=0x3F,
        flags=0x0,
        OS=0x40004,          # VOS_NT_WINDOWS32
        fileType=0x1,        # VFT_APP
        subtype=0x0,
        date=(0, 0),
    ),
    kids=[
        StringFileInfo([
            StringTable(
                "040904B0",  # US English, Unicode
                [
                    StringStruct("CompanyName", "The PDF Teleporter contributors"),
                    StringStruct("FileDescription",
                                 "PDF Teleporter command-line tool"),
                    StringStruct("FileVersion", "2.5.0"),
                    StringStruct("InternalName", "psditool"),
                    StringStruct("LegalCopyright",
                                 "Copyright (C) 2026 The PDF Teleporter contributors. "
                                 "Based on PDF Teleporter by F1GBD. GNU GPL v3."),
                    StringStruct("OriginalFilename", "psditool.exe"),
                    StringStruct("ProductName", "PDF Teleporter"),
                    StringStruct("ProductVersion", "2.5.0"),
                ],
            )
        ]),
        VarFileInfo([VarStruct("Translation", [0x0409, 1200])]),
    ],
)
